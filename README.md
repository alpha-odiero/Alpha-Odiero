  schedule:
    - cron: "0 0 * * *"
  workflow_dispatch:
  push:
    branches:
      - main

permissions:
  contents: write

  generate:
    runs-on: ubuntu-latest
    timeout-minutes: 5

    steps:
      - name: Generate snake animation
        uses: Platane/snk@v3

          target_branch: output
          build_dir: dist

          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
